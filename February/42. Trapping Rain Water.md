class Solution {
    public int trap(int[] height) {
        int n=height.length;
        int[]prefixsum=new int[n];
        int[]sufixsum=new int[n];
        prefixsum[0]=height[0];
        sufixsum[n-1]=height[n-1];
        for(int i=1;i<n;i++){
            if(height[i]>prefixsum[i-1]){
                prefixsum[i]=height[i];
            }
            else{
                prefixsum[i]=prefixsum[i-1];
            }
        }
        for(int i=n-2;i>0;i--){
            if(height[i]>sufixsum[i+1]){
                sufixsum[i]=height[i];
            }
            else{
                sufixsum[i]=sufixsum[i+1];
            }
        }
        int count=0;
        for(int i=1;i<n-1;i++){
            if(height[i]<prefixsum[i]&&height[i]<sufixsum[i]){
                count+=Math.min(prefixsum[i],sufixsum[i])-height[i];
            }
        }
        return(count);
        
    }
}
